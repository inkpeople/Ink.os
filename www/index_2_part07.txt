  setTimeout(()=>input && input.focus(),80);
  input.oninput=()=>runGlobalSearch(input.value.trim().toLowerCase());
  runGlobalSearch('');
}
function runGlobalSearch(q){
  const box=$('#globalSearchResults'); if(!box) return;
  if(!q){ box.innerHTML='<p class="dim small">Начните вводить запрос...</p>'; return; }
  const sections=[];
  const clientMatches=S.clients.filter(c=>[c.name,c.phone,c.instagram,c.telegram,c.city].filter(Boolean).some(v=>String(v).toLowerCase().includes(q))).slice(0,6);
  if(clientMatches.length) sections.push({title:'Клиенты',rows:clientMatches.map(c=>({label:c.name,sub:c.phone||c.city||'',onclick:`closeSheet();openClient('${c.id}')`}))});
  const eventMatches=S.events.filter(e=>[e.title,clientOrLabel(e),EVENT_TYPES[e.type]?.label].filter(Boolean).some(v=>String(v).toLowerCase().includes(q))).slice(0,6);
  if(eventMatches.length) sections.push({title:'Записи',rows:eventMatches.map(e=>({label:`${EVENT_TYPES[e.type]?.label||''} · ${clientOrLabel(e)}`,sub:fmtDate(e.date)+' '+(e.start||''),onclick:`closeSheet();go('calendar');openDaySummary('${e.date}')`}))});
  const projMatches=S.projects.filter(p=>[p.title,p.zone,(S.clients.find(c=>c.id===p.clientId)||{}).name].filter(Boolean).some(v=>String(v).toLowerCase().includes(q))).slice(0,6);
  if(projMatches.length) sections.push({title:'Проекты',rows:projMatches.map(p=>({label:p.title,sub:(S.clients.find(c=>c.id===p.clientId)||{}).name||'',onclick:`closeSheet();openProject('${p.id}')`}))});
  const finMatches=S.finance.filter(f=>[f.comment,INCOME_CATEGORIES[f.category],EXPENSE_CATEGORIES[f.category]].filter(Boolean).some(v=>String(v).toLowerCase().includes(q))).slice(0,6);
  if(finMatches.length) sections.push({title:'Финансы',rows:finMatches.map(f=>({label:(INCOME_CATEGORIES[f.category]||EXPENSE_CATEGORIES[f.category]||f.category||''),sub:fmtDate(f.date)+' · '+fmtMoney(f.amount),onclick:`closeSheet();go('finance')`}))});
  const invMatches=S.inventory.filter(i=>[i.name,i.config,i.brand].filter(Boolean).some(v=>String(v).toLowerCase().includes(q))).slice(0,6);
  if(invMatches.length) sections.push({title:'Материалы',rows:invMatches.map(i=>({label:i.name+(i.config?(' '+i.config):''),sub:`Остаток: ${i.qty} ${i.unit||'шт.'}`,onclick:`closeSheet();go('inventory')`}))});
  if(!sections.length){ box.innerHTML='<p class="dim small">Ничего не найдено.</p>'; return; }
  box.innerHTML=sections.map(s=>`
    <div class="small dim" style="margin:10px 0 4px;text-transform:uppercase;letter-spacing:.4px">${s.title}</div>
    ${s.rows.map(r=>`<div class="card" style="margin-bottom:6px;cursor:pointer" onclick="${r.onclick}"><b class="small">${esc(r.label)}</b><div class="small dim">${esc(r.sub)}</div></div>`).join('')}
  `).join('');
}
