function renderPurchaseRecommendation(){
  const lead=Number(S.settings.leadTimeDays||7), reserve=Number(S.settings.reserve||3);
  const now=Date.now();
  const cutoff=now-30*86400000;
  const usage30={};
  S.cartridgeUsage.filter(u=>new Date(u.date).getTime()>=cutoff).forEach(u=>{ const k=cartridgeKey(u); usage30[k]=(usage30[k]||0)+Number(u.qty||0); });
  const cartridgeItems=invItemsByCat('cartridge');
  const rows=cartridgeItems.map(item=>{
    const k=cartridgeKey({config:item.config,size:item.size});
    const used30=usage30[k]||0;
    const hasHistory=used30>0;
    const avgDay=used30/30;
    const recommended = hasHistory ? Math.ceil(avgDay*lead+reserve) : null;
    const toBuy = hasHistory ? Math.max(0,recommended-item.qty) : null;
    const status = !hasHistory ? 'недостаточно истории' : (item.qty<=item.threshold?'докупить срочно':(toBuy>0?'докупить':'достаточно'));
    return {name:(item.config||'')+' '+(item.size||''),qty:item.qty,used30,avgDay,threshold:item.threshold,toBuy,status,hasHistory};
  });
  $('#invBody').innerHTML=`
  <div class="card">
    <h3>Рекомендация закупки картриджей</h3>
    <p class="small dim">Формула: рекомендуемый запас = средний расход в день × срок поставки (${lead} дн.) + резерв (${reserve} шт.). Если истории мало — используется минимальный остаток, и предлагается ввести количество вручную.</p>
    <div class="table-wrap"><table><tr><th>Конфигурация</th><th>Остаток</th><th>30 дн.</th><th>Средн./день</th><th>Мин.</th><th>Купить</th><th>Статус</th></tr>
    ${rows.length?rows.map(r=>`<tr style="${r.status.includes('срочно')?'background:rgba(198,43,40,.1)':''}">
      <td>${esc(r.name)}</td><td>${r.qty}</td><td>${r.used30}</td><td>${r.avgDay.toFixed(2)}</td><td>${r.threshold}</td>
      <td>${r.hasHistory? r.toBuy : '<span class="dim">вручную</span>'}</td><td>${r.status}</td></tr>`).join(''):'<tr><td colspan="7" class="dim">Нет картриджей на складе. Добавьте позиции в разделе «Картриджи».</td></tr>'}

}

RENDERERS.cities=function(){
  const agg={};
  S.clients.forEach(c=>{
    if(!c.city) return;
    agg[c.city]=agg[c.city]||{clients:0,revenue:0,visits:0};
    agg[c.city].clients++;
    agg[c.city].revenue+=Number(c.totalSum||0);
    agg[c.city].visits+=Number(c.visits||0);

  const totalRevenue=Object.values(agg).reduce((s,c)=>s+c.revenue,0)||1;

  const startBtn=$('#camStartBtn');
  if(startBtn) startBtn.onclick=async ()=>{

function stopCamera(){ if(photoState.cameraStream){ photoState.cameraStream.getTracks().forEach(t=>t.stop()); photoState.cameraStream=null; } }
function analyzeFrameLoop(){
  const v=$('#camVideo'); const ov=$('#camOverlay');
  if(!v || v.style.display==='none' || !photoState.cameraStream) return;
  ov.width=v.clientWidth; ov.height=v.clientHeight;
  const tmp=document.createElement('canvas'); tmp.width=80; tmp.height=100;
  const tctx=tmp.getContext('2d'); tctx.drawImage(v,0,0,80,100);
  let data; try{ data=tctx.getImageData(0,0,80,100).data; }catch(e){ return; }
  let sum=0,count=0,edgeSum=0;
  for(let i=0;i<data.length;i+=4){ const l=(data[i]+data[i+1]+data[i+2])/3; sum+=l; count++; }
  const avg=sum/count; // brightness 0-255
  // crude sharpness via variance of luminance diff between neighboring samples
  let variance=0,prev=null,n=0;
  for(let i=0;i<data.length;i+=16){ const l=(data[i]+data[i+1]+data[i+2])/3; if(prev!==null){ variance+=Math.abs(l-prev); n++; } prev=l; }
  const sharpness=n?variance/n:0;
  let score=100, hints=[];
  if(avg<60){ score-=30; hints.push('Слишком темно — измените свет'); }
  else if(avg>200){ score-=25; hints.push('Пересвет — уберите блик или уменьшите свет'); }
  if(sharpness<3){ score-=25; hints.push('Возможна дрожь камеры — держите ровно'); }
  if(hints.length===0){ hints.push('Кадр выглядит стабильным'); }
  score=Math.max(0,Math.min(100,score));
  const frameEl=$('#cameraFrame');
  frameEl.className='camera-frame '+(score>70?'good':score>40?'warn':'bad');
  $('#camScoreWrap').innerHTML=`<div>Оценка кадра: <b>${score}/100</b></div><div class="dim">${hints.join(' · ')}</div>`;
  requestAnimationFrame(analyzeFrameLoop);
}

function applyPreset(key){
  const p=PHOTO_PRESETS[key];
  if($('#rBrightness')) $('#rBrightness').value=(p.brightness||1)*100;
  if($('#rContrast')) $('#rContrast').value=(p.contrast||1)*100;
  if($('#rSaturation')) $('#rSaturation').value=(p.saturation||1)*100;
  renderEditCanvas();
}
function renderEditCanvas(){
  if(!photoState.img) return;
  const canvas=$('#editCanvas'); const img=photoState.img;
  let w=img.width,h=img.height; const maxW=640; if(w>maxW){ h=h*maxW/w; w=maxW; }
  canvas.width=w; canvas.height=h;
  const ctx=canvas.getContext('2d');
  const b=($('#rBrightness')?.value||100)/100, c=($('#rContrast')?.value||100)/100, s=($('#rSaturation')?.value||100)/100, r=($('#rRedReduce')?.value||0)/100;
  ctx.filter=`brightness(${b}) contrast(${c}) saturate(${s})`;
  ctx.drawImage(img,0,0,w,h);
  ctx.filter='none';
  if(r>0){
    const id=ctx.getImageData(0,0,w,h); const d=id.data;
    for(let i=0;i<d.length;i+=4){
      const avg=(d[i]+d[i+1]+d[i+2])/3;
      if(d[i]>d[i+1]+18 && d[i]>d[i+2]+18){ d[i]=d[i]-(d[i]-avg)*r; }
    }
    ctx.putImageData(id,0,0);
  }
  if($('#cLogo')?.checked && S.settings.logo){
    const logo=new Image();
    logo.onload=()=>{ const lw=w*0.22; const lh=logo.height*(lw/logo.width); ctx.globalAlpha=0.85; ctx.drawImage(logo,w-lw-10,h-lh-10,lw,lh); ctx.globalAlpha=1; };
    logo.src=S.settings.logo;
  }
}

function generateWeeklyReport(){
  const now=new Date(); const start=new Date(now); start.setDate(now.getDate()-now.getDay()+1);
  const s=start.toISOString().slice(0,10);
  const weekEvents=S.events.filter(e=>e.date>=s);
  const sessions=weekEvents.filter(e=>e.type==='session').length;
  const consults=weekEvents.filter(e=>e.type==='consult').length;
  const corrections=weekEvents.filter(e=>e.type==='correction').length;
  const models=weekEvents.filter(e=>e.type==='model').length;
  const newClients=S.clients.filter(c=>c.firstVisit>=s).length;
  const revenue=S.finance.filter(f=>f.type==='income'&&f.date>=s).reduce((a,f)=>a+Number(f.amount||0),0);
  const expenses=S.finance.filter(f=>f.type==='expense'&&f.date>=s).reduce((a,f)=>a+Number(f.amount||0),0);
  const filmSales=S.healingFilmSales.filter(f=>f.date>=s).reduce((a,f)=>a+Number(f.amount||0),0);
  const text=`Отчёт за неделю от ${fmtDate(s)}\n\nСеансы: ${sessions}\nКонсультации: ${consults}\nКоррекции: ${corrections}\nМодели: ${models}\nНовые клиенты: ${newClients}\nВыручка: ${fmtMoney(revenue)}\nРасходы: ${fmtMoney(expenses)}\nПродажи плёнки: ${fmtMoney(filmSales)}\nЧистая прибыль (предварительно): ${fmtMoney(revenue-expenses)}`;
  const rev={id:uid(),date:todayISO(),weekStart:s,text};
  idbPut('reminders',rev);
  window._lastReportText=text;
  window._lastReportSeries=[
    {label:'Сеансы',value:sessions,color:'#20B2AA'},
    {label:'Консульт.',value:consults,color:'#F4C542'},
    {label:'Коррекции',value:corrections,color:'#E53935'},
    {label:'Модели',value:models,color:'#8E44AD'},
    {label:'Новые кл.',value:newClients,color:'#60A5FA'}
  ];
  $('#reportOut').innerHTML=`<div class="card" style="margin-bottom:10px"><canvas id="reportChartCanvas" style="width:100%;height:180px"></canvas></div>
  <div class="card"><pre style="white-space:pre-wrap;font-family:inherit">${esc(text)}</pre>
  <div class="row" style="margin-top:8px"><button class="btn btn-sm" onclick="copyLastReport()">Копировать</button></div></div>`;
  setTimeout(()=>drawBarChart($('#reportChartCanvas'),window._lastReportSeries),20);
}

}

function downloadBlob(blob,filename){
  const url=URL.createObjectURL(blob); const a=document.createElement('a'); a.href=url; a.download=filename; document.body.appendChild(a); a.click(); a.remove(); setTimeout(()=>URL.revokeObjectURL(url),2000);

function exportCSV(store){
  const rows=S[store]||[];
  if(!rows.length){ toast('Нет данных для экспорта'); return; }
  const keys=Array.from(new Set(rows.flatMap(r=>Object.keys(r))));
  const csv=[keys.join(',')].concat(rows.map(r=>keys.map(k=>`"${String(r[k]==null?'':r[k]).replace(/"/g,'""')}"`).join(','))).join('\n');
  downloadBlob(new Blob([csv],{type:'text/csv'}),`${store}_${todayISO()}.csv`);
}
function startImport(mode){
  const f=$('#importFile').files[0];
  if(!f){ toast('Выберите файл для импорта'); return; }
  const reader=new FileReader();
  reader.onload=async ev=>{
    let json; try{ json=JSON.parse(ev.target.result); }catch(e){ toast('Файл повреждён или не JSON'); return; }
    const counts=Object.entries(json).map(([k,v])=>`${k}: ${Array.isArray(v)?v.length:1}`).join(', ');
    confirmDialog(`Найдено: ${counts}. Создать резервную копию перед импортом и продолжить (${mode==='replace'?'ЗАМЕНА':'объединение'})?`, async ()=>{
      exportFullBackup();
      for(const [store,items] of Object.entries(json)){
        if(!STORES.includes(store)) continue;
        if(store==='settings'){ S.settings={...S.settings,...items}; await saveSettings(); continue; }
        if(mode==='replace'){ S[store]=[]; }
        const arr=Array.isArray(items)?items:[items];
        for(const it of arr){
          const idx=S[store].findIndex(x=>x.id===it.id);
          if(idx>=0) S[store][idx]=it; else S[store].push(it);
          await idbPut(store,it);
        }
      }
      toast('Импорт завершён'); go('data');
    });
  };
  reader.readAsText(f);
}

const CALC_SIZES={'Маленькая (до 5 см)':1,'Средняя (5-15 см)':1.6,'Крупная (15-25 см)':2.4,'Очень крупная (25+ см)':3.4};
const CALC_COMPLEXITY={'Простая':1,'Средняя':1.3,'Сложная':1.7,'Очень сложная':2.2};
RENDERERS.calculator=function(){
  const cur=S.settings.calcCurrency||'₽';
  $('#content').innerHTML=`
  <h2>Калькулятор цены</h2>
  <div class="card">
    <form id="calcForm">
    <div class="field-row">
      <div><label>Почасовая ставка</label><input type="number" name="rate" value="${S.settings.calcRate||3000}"></div>
      <div><label>Валюта</label><select name="currency"><option ${cur==='₽'?'selected':''}>₽</option><option ${cur==='$'?'selected':''}>$</option><option ${cur==='€'?'selected':''}>€</option><option ${cur==='₸'?'selected':''}>₸</option></select></div>
    </div>
    <div class="field-row">
      <div><label>Минимальная стоимость</label><input type="number" name="minCost" value="1500"></div>
      <div><label>Часы (предположительно)</label><input type="number" name="hours" value="3" step="0.5"></div>
    </div>
    <label>Размер</label><select name="size">${Object.keys(CALC_SIZES).map(k=>`<option>${k}</option>`).join('')}</select>
    <label>Сложность</label><select name="complexity">${Object.keys(CALC_COMPLEXITY).map(k=>`<option>${k}</option>`).join('')}</select>
    <div class="field-row">
      <div><label>Цвет</label><select name="colorType"><option>Чёрно-белая</option><option>Цветная</option></select></div>
      <div><label>Количество сеансов</label><input type="number" name="sessions" value="1" min="1"></div>
    </div>
    <div class="field-row">
      <div><label>Выездная наценка</label><input type="number" name="travelFee" value="0"></div>
      <div><label>Скидка, %</label><input type="number" name="discount" value="0"></div>
    </div>
    <div class="field-row">
      <div><label>Стоимость расходников</label><input type="number" name="materialsCost" value="0"></div>
      <div><label>Предоплата</label><input type="number" name="prepay" value="0"></div>
    </div>
    </form>
  </div>
  <div class="card" style="margin-top:12px" id="calcOut"></div>
  <p class="small dim" style="margin-top:8px">Итоговую цену внизу можно поправить вручную — калькулятор запомнит вашу правку и не будет пересчитывать её, пока вы не измените параметры выше.</p>`;
  let manualTotal=null;
  function recalc(){
    const fd=new FormData($('#calcForm'));
    const rate=Number(fd.get('rate')||0), minCost=Number(fd.get('minCost')||0), hours=Number(fd.get('hours')||0);
    const sizeMult=CALC_SIZES[fd.get('size')]||1, complexMult=CALC_COMPLEXITY[fd.get('complexity')]||1;
    const colorMult=fd.get('colorType')==='Цветная'?1.25:1;
    const sessions=Math.max(1,Number(fd.get('sessions')||1));
    const travelFee=Number(fd.get('travelFee')||0), discountPct=Number(fd.get('discount')||0);
    const materialsCost=Number(fd.get('materialsCost')||0), prepay=Number(fd.get('prepay')||0);
    S.settings.calcRate=rate; S.settings.calcCurrency=fd.get('currency'); saveSettings();
    const curNow=fd.get('currency');
    const base=rate*hours;
    const withCoeffs=base*sizeMult*complexMult*colorMult;
    const discountAmt=withCoeffs*discountPct/100;
    let total=withCoeffs - discountAmt + travelFee + materialsCost;
    total=Math.max(minCost,total);
    const finalTotal = manualTotal!=null ? manualTotal : total;
    const remain=finalTotal-prepay;
    const perSession=finalTotal/sessions;
    $('#calcOut').innerHTML=`
    <div class="kv"><span>Базовая стоимость (ставка × часы)</span><span>${base.toLocaleString('ru-RU')} ${curNow}</span></div>
    <div class="kv"><span>С коэффициентами (размер×сложность×цвет)</span><span>${withCoeffs.toLocaleString('ru-RU')} ${curNow}</span></div>
    <div class="kv"><span>Скидка</span><span>-${discountAmt.toLocaleString('ru-RU')} ${curNow}</span></div>
    <div class="kv"><span>Выездная наценка</span><span>+${travelFee.toLocaleString('ru-RU')} ${curNow}</span></div>
    <div class="kv"><span>Стоимость расходников</span><span>+${materialsCost.toLocaleString('ru-RU')} ${curNow}</span></div>
    <div class="kv" style="border-top:1px solid var(--line-red);margin-top:6px;padding-top:10px">
      <span><b>Итоговая стоимость</b></span>
      <span><input type="number" id="calcManualTotal" value="${finalTotal.toFixed(0)}" style="width:120px;display:inline-block;text-align:right;color:var(--red-soft);font-weight:800"> ${curNow}</span>
    </div>
    <div class="kv"><span>Предоплата</span><span>${prepay.toLocaleString('ru-RU')} ${curNow}</span></div>
    <div class="kv"><span>Остаток</span><span style="color:var(--red-soft);font-weight:700">${remain.toLocaleString('ru-RU')} ${curNow}</span></div>
    <div class="kv"><span>Стоимость одного сеанса (из ${sessions})</span><span>${perSession.toLocaleString('ru-RU')} ${curNow}</span></div>
    `;
    $('#calcManualTotal').oninput=e=>{ manualTotal=Number(e.target.value)||0; recalc(); };
  }
  $$('#calcForm input, #calcForm select').forEach(el=>el.addEventListener('input',()=>{ manualTotal=null; recalc(); }));
  recalc();
};
