function downloadBlob(blob,filename){
  const url=URL.createObjectURL(blob); const a=document.createElement('a'); a.href=url; a.download=filename; document.body.appendChild(a); a.click(); a.remove(); setTimeout(()=>URL.revokeObjectURL(url),2000);
}
function exportCSV(store){
  const rows=S[store]||[];
  if(!rows.length){ toast('Нет данных для экспорта'); return; }
  const keys=Array.from(new Set(rows.flatMap(r=>Object.keys(r))));
  const csv=[keys.join(',')].concat(rows.map(r=>keys.map(k=>`"${String(r[k]==null?'':r[k]).replace(/"/g,'""')}"`).join(','))).join('\n');
  downloadBlob(new Blob([csv],{type:'text/csv'}),`${store}_${todayISO()}.csv`);
}
function startImport(mode){
  const f=$('#importFile').files[0];
  if(!f){ toast('Выберите файл для импорта'); return; }
  const reader=new FileReader();
  reader.onload=async ev=>{
    let json; try{ json=JSON.parse(ev.target.result); }catch(e){ toast('Файл повреждён или не JSON'); return; }
    const counts=Object.entries(json).map(([k,v])=>`${k}: ${Array.isArray(v)?v.length:1}`).join(', ');
    confirmDialog(`Найдено: ${counts}. Создать резервную копию перед импортом и продолжить (${mode==='replace'?'ЗАМЕНА':'объединение'})?`, async ()=>{
      exportFullBackup();
      for(const [store,items] of Object.entries(json)){
        if(!STORES.includes(store)) continue;
        if(store==='settings'){ S.settings={...S.settings,...items}; await saveSettings(); continue; }
        if(mode==='replace'){ S[store]=[]; }
        const arr=Array.isArray(items)?items:[items];
        for(const it of arr){
          const idx=S[store].findIndex(x=>x.id===it.id);
          if(idx>=0) S[store][idx]=it; else S[store].push(it);
          await idbPut(store,it);
        }
      }
      try{ normalizeData(); }catch(e){ console.error(e); }
      toast('Импорт завершён'); go('data');
    });
  };
  reader.readAsText(f);
}

function anStats(f,t){
  const fin=(S.finance||[]).filter(x=>x.date>=f&&x.date<=t), sm=a=>a.reduce((n,x)=>n+Number(x.amount||0),0);

  const ev=S.events.filter(e=>e.status==='completed'&&e.date>=f&&e.date<=t);
  return {inc,exp,clients:new Set(ev.map(e=>e.clientId).filter(Boolean)).size,avg:ev.length?inc/ev.length:0,net:inc-exp-inc*Number(S.settings.taxRate||0)/100};
}

function pushCandidates(hz){
  hz=hz||70*3600000; const cfg=notifCfg(), pc=Object.assign({},pushCfg(),{names:pushCfg().names||isNative()}), now=Date.now(), horizon=now+hz, out=[];
  const put=(ref,when,title,text)=>{ if(when>now+20000&&when<=horizon) out.push({ref,at:Math.floor(when/1000),title,text}); };
  const D=(date,h)=>new Date(date+'T'+String(h).padStart(2,'0')+':00').getTime();
  const days=Array.from({length:Math.min(32,Math.ceil(hz/86400000)+1)},(_,n)=>isoL(new Date(now+n*86400000)));
  S.events.filter(e=>e.date&&NOTIFY_TYPES.includes(e.type)&&!['cancelled','completed','noshow'].includes(e.status)).forEach(e=>{
    const start=new Date(e.date+'T'+(e.start||'10:00')).getTime(), c=e.clientId&&S.clients.find(x=>x.id===e.clientId), tl=EVENT_TYPES[e.type].label;

    if(cfg.dayOf){ const sh=+(e.start||'10:00').slice(0,2); put('d-'+key,sh>=9?D(e.date,8):start-3600000,'Сегодня',`В ${e.start||''}: ${tl}${tail}`); }
  });
  const low=S.inventory.filter(i=>i.category!=='equipment'&&Number(i.qty)<=Number(i.threshold));
  if(cfg.restock&&low.length){ const t=low.slice(0,3).map(i=>i.name).join(', ')+(low.length>3?' и ещё '+(low.length-3):''); days.forEach(d=>put('restock-'+d,D(d,10),'Пополнение склада',`Нужно закупить ${low.length} поз.: ${t}.`)); }
  if(cfg.expenses) days.forEach(d=>{ if(S.events.some(e=>e.date===d&&['session','consult','correction','model'].includes(e.type)&&e.status!=='cancelled')&&!(S.finance||[]).some(f=>f.type!=='income'&&f.date===d)) put('exp-'+d,D(d,19),'Расходы','Не забудьте внести расходы за сегодня.'); });
  if(cfg.write) S.events.filter(e=>e.type==='session'&&e.status==='completed'&&e.clientId).forEach(e=>{ const c=S.clients.find(x=>x.id===e.clientId); if(!c) return; const who=pc.names?' '+c.name:' клиенту';
    if(!e.careSent){ const nx=isoL(new Date(new Date(e.date+'T12:00').getTime()+86400000)); put('care-'+e.id,[D(e.date,12),D(nx,12)].find(t=>t>now)||0,'Уход',`Отправьте${who} инструкцию по уходу.`); }
    if(e.healCheck) put(`heal-${e.id}-${e.healCheck}`,D(e.healCheck,10),'Заживление',`Напишите${who}: узнайте, как заживает, и попросите фото.`); });
  return out;
}

function earnCfg(){ return Object.assign({desired:0,days:5,avg:0,fixed:[]},S.settings.earn||{}); }
function earnPlan(){
  const E=earnCfg(), tax=Number(S.settings.taxRate||0), now=new Date(), ym=isoL(now).slice(0,7);
  const fixedSum=E.fixed.reduce((a,f)=>a+Number(f.amount||0),0), des=Number(E.desired||0);
  const required=(des>0||fixedSum>0)?Math.ceil((des+fixedSum)/(1-Math.min(tax,95)/100)):Number(S.settings.monthGoal||0);
  const from=isoL(new Date(now.getTime()-90*86400000)), ev=S.events.filter(e=>e.status==='completed'&&Number(e.cost)>0&&e.date>=from);
  const auto=ev.length?Math.round(ev.reduce((a,e)=>a+Number(e.cost),0)/ev.length):0, avg=Number(E.avg||0)||auto;
  const income=(S.finance||[]).filter(f=>f.type==='income'&&(f.date||'').startsWith(ym)).reduce((a,f)=>a+Number(f.amount||0),0);
  const dim=new Date(now.getFullYear(),now.getMonth()+1,0).getDate(), wdLeft=Math.max(1,Math.ceil((dim-now.getDate()+1)*E.days/7)), wdMonth=Math.max(1,Math.round(dim*E.days/7));
  const left=Math.max(0,required-income);
  return {E,tax,fixedSum,des,required,avg,auto,income,left,pct:required?Math.min(100,Math.round(income/required*100)):0,sessions:avg?Math.ceil(required/avg):0,sessionsLeft:avg?Math.ceil(left/avg):0,perDay:Math.ceil(required/wdMonth),perDayLeft:Math.ceil(left/wdLeft),perWeek:avg?Math.ceil(required/avg/(dim/7)):0};
}
