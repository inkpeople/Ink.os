/* ====================== GLOBAL DOUBLE-SUBMIT GUARD ======================
   Rapid double-tap on a submit button (common on touch devices, especially
   with any perceived lag) used to create duplicate records, since none of
   the ~18 forms in this app had their own submit lock. Fixed once, globally,
   on the capture phase — runs before the form's own onsubmit handler. */
document.addEventListener('submit', function(e){
  const f=e.target;
  if(!(f instanceof HTMLFormElement)) return;
  const now=Date.now();
  if(f._lastSubmitAt && now-f._lastSubmitAt<1000){
    e.preventDefault(); e.stopImmediatePropagation();
    return;
  }
  f._lastSubmitAt=now;
}, true);

// Shows a toast with an "Отменить" button. Item stays in IndexedDB for `duration` ms
// (removed only from the in-memory list + UI), so undo is instant and lossless.
function showUndoToast(msg,onUndo,duration){
  duration=duration||5000;
  const w=$('#toastWrap');
  const el=document.createElement('div');
  el.className='toast'; el.style.display='flex'; el.style.alignItems='center'; el.style.gap='10px';
  const span=document.createElement('span'); span.textContent=msg; span.style.flex='1';
  const btn=document.createElement('button'); btn.textContent='Отменить'; btn.className='btn btn-sm btn-primary'; btn.style.flexShrink='0';
  el.appendChild(span); el.appendChild(btn); w.appendChild(el);
  let done=false;
  const timer=setTimeout(()=>{ if(!done){ done=true; el.style.opacity='0'; el.style.transition='opacity .4s'; setTimeout(()=>el.remove(),400); } },duration);
  btn.onclick=()=>{ if(done) return; done=true; clearTimeout(timer); el.remove(); onUndo&&onUndo(); };
}
// opts: {store, listKey, id, message, onChange, auditEntity, auditLabel}
function trashDelete(opts){
  const list=S[opts.listKey];
  const idx=list.findIndex(x=>x.id===opts.id);
  if(idx<0) return;
  const item=list[idx];
  list.splice(idx,1);
  if(opts.onChange) opts.onChange();
  let undone=false;
  const timer=setTimeout(async()=>{
    if(!undone){
      await idbDel(opts.store,opts.id);
      if(opts.auditEntity) logAudit('delete',opts.auditEntity,opts.auditLabel||'',`Удалено окончательно`);
    }
  },5000);
  showUndoToast(opts.message||'Удалено.',()=>{
    undone=true; clearTimeout(timer);
    list.splice(idx,0,item);
    if(opts.onChange) opts.onChange();
    toast('Восстановлено');
  });
}

function drawBarChart(canvas,series,opts){
  opts=opts||{};
  const ctx=canvas.getContext('2d');
  const w=canvas.width=canvas.clientWidth||300, h=canvas.height=opts.height||180;
  ctx.clearRect(0,0,w,h);
  const pad=28;
  const max=Math.max(1,...series.map(s=>s.value));
  const barW=(w-pad*2)/series.length;
  ctx.strokeStyle='#2a2523'; ctx.beginPath(); ctx.moveTo(pad,h-24); ctx.lineTo(w-6,h-24); ctx.stroke();
  series.forEach((s,i)=>{
    const bh=(h-50)*(s.value/max);
    const x=pad+i*barW+barW*0.15, y=h-24-bh, bw=barW*0.7;
    ctx.fillStyle=s.color||'#c62b28';
    ctx.fillRect(x,y,bw,bh);
    ctx.fillStyle='#a89e98'; ctx.font='10px sans-serif'; ctx.textAlign='center';
    ctx.fillText(s.label,x+bw/2,h-10);
    ctx.fillStyle='#eee8e4'; ctx.fillText(String(s.value),x+bw/2,y-4>10?y-4:10);
  });
}
function drawLineChart(canvas,points,opts){
  opts=opts||{};
  const ctx=canvas.getContext('2d');
  const dpr=window.devicePixelRatio||1;
  const w=canvas.clientWidth||300, h=opts.height||180;
  canvas.width=w*dpr; canvas.height=h*dpr; ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.clearRect(0,0,w,h);
  if(!points.length) return;
  const padL=42,padR=10,padT=14,padB=22;
  const max=Math.max(1,...points.map(p=>p.value));
  const min=Math.min(0,...points.map(p=>p.value));
  const plotW=w-padL-padR, plotH=h-padT-padB;
  // gridlines + y labels
  ctx.strokeStyle='#241c1c'; ctx.fillStyle='#6f655f'; ctx.font='9px sans-serif'; ctx.textAlign='right';
  const steps=4;
  for(let i=0;i<=steps;i++){
    const val=min+(max-min)*i/steps;
    const y=padT+plotH-(plotH*i/steps);
    ctx.beginPath(); ctx.moveTo(padL,y); ctx.lineTo(w-padR,y); ctx.stroke();
    ctx.fillText(Math.round(val).toLocaleString('ru-RU'),padL-6,y+3);
  }
  const stepX=points.length>1?plotW/(points.length-1):0;
  const xy=points.map((p,i)=>({x:padL+i*stepX,y:padT+plotH-(max===min?0:plotH*(p.value-min)/(max-min))}));
  // filled area
  const grad=ctx.createLinearGradient(0,padT,0,padT+plotH);
  grad.addColorStop(0,'rgba(198,43,40,.35)'); grad.addColorStop(1,'rgba(198,43,40,0)');
  ctx.beginPath(); ctx.moveTo(xy[0].x,padT+plotH);
  xy.forEach(p=>ctx.lineTo(p.x,p.y));
  ctx.lineTo(xy[xy.length-1].x,padT+plotH); ctx.closePath();
  ctx.fillStyle=grad; ctx.fill();
  // line
  ctx.beginPath(); ctx.moveTo(xy[0].x,xy[0].y);
  xy.forEach(p=>ctx.lineTo(p.x,p.y));
  ctx.strokeStyle='#e5453f'; ctx.lineWidth=2; ctx.shadowColor='rgba(198,43,40,.6)'; ctx.shadowBlur=6; ctx.stroke(); ctx.shadowBlur=0;
  // dots
  ctx.fillStyle='#e5453f';
  xy.forEach(p=>{ ctx.beginPath(); ctx.arc(p.x,p.y,2.6,0,7); ctx.fill(); });
  // x labels (skip some if crowded)
  ctx.fillStyle='#6f655f'; ctx.textAlign='center'; ctx.font='9px sans-serif';
  const labelEvery=Math.max(1,Math.ceil(points.length/6));
  points.forEach((p,i)=>{ if(i%labelEvery===0||i===points.length-1) ctx.fillText(p.label,padL+i*stepX,h-6); });
}

function idbOpen(){
  return new Promise((resolve,reject)=>{
    if(!('indexedDB' in window)){ reject('no-idb'); return; }
    const req=indexedDB.open(DB_NAME,DB_VER);
    req.onupgradeneeded=e=>{
      const d=e.target.result;
      STORES.forEach(name=>{
        if(!d.objectStoreNames.contains(name)){
          const store=d.createObjectStore(name,{keyPath:'id'});
          if(name!=='settings') store.createIndex('byUpdated','updatedAt',{unique:false});
        }
      });
    };
    req.onsuccess=e=>resolve(e.target.result);
    req.onerror=e=>reject(e);
  });
}
function idbAll(store){
  return new Promise((resolve,reject)=>{
    if(!db){ resolve(S[store]||[]); return; }
    const tx=db.transaction(store,'readonly');
    const req=tx.objectStore(store).getAll();
    req.onsuccess=()=>resolve(req.result||[]);
    req.onerror=()=>reject(req.error);
  });
}
function idbPut(store,obj){
  obj.updatedAt=Date.now();
  if(!obj.id) obj.id=uid();
  return new Promise((resolve,reject)=>{
    if(!db){ // localStorage fallback
      const arr=S[store]||[]; const i=arr.findIndex(x=>x.id===obj.id);
      if(i>=0) arr[i]=obj; else arr.push(obj);
      S[store]=arr; persistLocalFallback();
      resolve(obj); return;
    }
    const tx=db.transaction(store,'readwrite');
    tx.objectStore(store).put(obj);
    tx.oncomplete=()=>resolve(obj);
    tx.onerror=()=>reject(tx.error);
  });
}
function idbDel(store,id){
  return new Promise((resolve,reject)=>{
    if(!db){ S[store]=(S[store]||[]).filter(x=>x.id!==id); persistLocalFallback(); resolve(); return; }
    const tx=db.transaction(store,'readwrite');
    tx.objectStore(store).delete(id);    tx.oncomplete=()=>resolve();
    tx.onerror=()=>reject(tx.error);
  });
}

function computeAutoNotifications(){
  const now=Date.now();
  const existingRefs=new Set(S.notifications.map(n=>n.refId));
  // post-session reminders (7/30/60 days) based on tattooWork/session events completed
  S.events.filter(e=>e.type==='session' && e.status==='completed').forEach(ev=>{
    const client=S.clients.find(c=>c.id===ev.clientId);
    if(!client) return;
    const d0=new Date(ev.date);
    [[7,'week',`Прошла неделя после сеанса с ${client.name}. Напишите клиенту и узнайте, как проходит восстановление.`],
     [30,'month',`Прошёл месяц после сеанса с ${client.name}. Узнайте, как выглядит татуировка, и предложите продолжение проекта.`],
     [60,'2month',`Прошло два месяца после сеанса с ${client.name}. Предложите новую идею и попросите оставить отзыв.`]
    ].forEach(([days,tag,text])=>{
      const refId=ev.id+'-'+tag;
      const due=new Date(d0.getTime()+days*86400000);
      if(due.getTime()<=now && !existingRefs.has(refId)){
        addNotification(text,'followup',refId);
      }
    });
  });
  // birthdays: warn 7 days ahead, and on the day itself, once per year
  (S.clients||[]).forEach(c=>{
    if(!c.birthDate) return;
    const bd=new Date(c.birthDate); if(isNaN(bd.getTime())) return;
    const year=new Date(now).getFullYear();
    [0,7].forEach(daysAhead=>{
      const target=new Date(year,bd.getMonth(),bd.getDate());
      target.setDate(target.getDate()-daysAhead);
      const refId=`bday-${c.id}-${year}-${daysAhead}`;
      const diff=Math.abs(target.getTime()-now);
      if(diff<86400000 && !existingRefs.has(refId)){
        addNotification(daysAhead===0?`Сегодня день рождения у ${c.name}! 🎂`:`Через неделю день рождения у ${c.name}.`,'birthday',refId);
      }
    });
  });
  // homework overdue
  S.homework.forEach(h=>{
    if(h.deadline && h.status!=='Принято' && h.status!=='Отменено'){
      if(new Date(h.deadline).getTime()<now && h.status!=='Просрочено'){
        h.status='Просрочено'; idbPut('homework',h);
        addNotification(`Домашнее задание просрочено: ${h.title||''} (${(S.students.find(s=>s.id===h.studentId)||{}).name||''})`,'homework','hwover-'+h.id);
      }
    }
  });
  // backup reminder: warn if there's meaningful data but no backup in the last 14 days
  const hasData = (S.clients||[]).length>0 || (S.events||[]).length>0;
  if(hasData){
    const lastBackupTs = (S.backups||[]).reduce((max,b)=>{ const t=new Date(b.date).getTime(); return t>max?t:max; },0);
    const daysSince = lastBackupTs? Math.floor((now-lastBackupTs)/86400000) : Infinity;
    if(daysSince>=14){
      const bucket = Number.isFinite(daysSince)? Math.floor(daysSince/7) : 'never';
      const refId='backup-reminder-'+bucket;
      if(!existingRefs.has(refId)){
        addNotification(lastBackupTs? `Вы не делали резервную копию ${daysSince} дн. Рекомендуем сохранить копию в разделе «Данные».` : `Вы ещё ни разу не делали резервную копию данных. Рекомендуем сохранить копию в разделе «Данные».`,'backup',refId);
      }
    }
  }
  renderNotifCount();
}

function runSplash(){
  const st=S.settings||{};
  if(st.splash==='off'){ finishSplash(); return; }
  const sp=$('#splash'), short=st.splash==='short';
  sp.classList.toggle('short',short);
  requestAnimationFrame(()=>sp.classList.add('play'));
  setTimeout(finishSplash, short?2900:5600);
}
function finishSplash(){
  if(finishSplash._done) return;
  finishSplash._done=true;
  const sp=$('#splash');
  sp.style.opacity='0';
  setTimeout(()=>{ sp.style.display='none'; afterSplash(); },600);
}
$('#skipSplash').onclick=finishSplash;

/* ====================== PIN LOCK ====================== */
let pinBuffer='';
function afterSplash(){
  if(S.settings.pin){
    showPinScreen('unlock');
  } else {
    bootApp();
  }
}
function showPinScreen(mode){
  $('#pinScreen').classList.remove('hidden');
  pinBuffer='';
  $('#pinMsg').textContent = mode==='set' ? 'Придумайте PIN (4 цифры)' : (mode==='confirm'?'Повторите PIN':'Введите PIN');
  renderPinDots();
  const digits=[1,2,3,4,5,6,7,8,9,'',0,'⌫'];
  $('#pinPad').innerHTML=digits.map(d=>`<button data-d="${d}" ${d===''?'style="visibility:hidden"':''}>${d}</button>`).join('');
  $('#pinPad').querySelectorAll('button[data-d]').forEach(b=>{
    b.onclick=()=>{
      const d=b.dataset.d;
      if(d==='⌫'){ pinBuffer=pinBuffer.slice(0,-1); renderPinDots(); return; }
      if(pinBuffer.length<4){ pinBuffer+=d; renderPinDots(); }
      if(pinBuffer.length===4) handlePinComplete(mode);
    };
  });
}
let pinFirstEntry=null;
function handlePinComplete(mode){
  if(mode==='unlock'){
    if(pinBuffer===S.settings.pin){ hidePinScreen(); bootApp(); }
    else { $('#pinMsg').textContent='Неверный PIN'; pinBuffer=''; renderPinDots(); }
  } else if(mode==='set'){
    pinFirstEntry=pinBuffer; showPinScreen('confirm');
  } else if(mode==='confirm'){
    if(pinBuffer===pinFirstEntry){ S.settings.pin=pinBuffer; saveSettings(); hidePinScreen(); toast('PIN установлен'); if(!appBooted) bootApp(); else go('settings'); }
    else { toast('PIN не совпадает'); showPinScreen('set'); }
  }
}
function renderPinDots(){
  $('#pinDots').innerHTML=[0,1,2,3].map(i=>`<span class="${i<pinBuffer.length?'filled':''}"></span>`).join('');
}
function hidePinScreen(){ $('#pinScreen').classList.add('hidden'); }
$('#lockBtn').onclick=()=>{ if(S.settings.pin){ showPinScreen('unlock'); } else { toast('PIN не установлен. Задайте его в Настройках.'); } };

/* ====================== AUTO-LOCK AFTER INACTIVITY ====================== */
let inactivityTimer=null;
function resetInactivityTimer(){
  if(inactivityTimer){ clearTimeout(inactivityTimer); inactivityTimer=null; }
  const mins=Number(S.settings.autoLockMinutes||0);
  if(!S.settings.pin || !mins) return;
  inactivityTimer=setTimeout(()=>{
    if(!$('#pinScreen').classList.contains('hidden')) return;
    showPinScreen('unlock');
  }, mins*60000);
}
function setupInactivityLock(){
  ['click','touchstart','keydown','mousemove'].forEach(evt=>document.addEventListener(evt,resetInactivityTimer,{passive:true}));
  resetInactivityTimer();
}

/* ====================== МАССОВАЯ РАССЫЛКА (WhatsApp, по одному клику на клиента) ====================== */
function openBulkMessageSheet(clients){
  clients=(clients||[]).filter(Boolean);
  if(!clients.length){ toast('Нет клиентов для рассылки'); return; }
  const defaultText='Здравствуйте! Давно не виделись — хотели узнать, как ваши дела, и предложить записаться на новый сеанс 🙂';
  openSheet('Массовая рассылка · '+clients.length+' чел.', `
    <p class="small dim">Один текст для всех. Нажмите WhatsApp у каждого клиента — сообщение откроется уже с текстом, останется нажать «Отправить».</p>
    <textarea id="bulkMsgText" rows="3">${esc(defaultText)}</textarea>
    <div class="row" style="margin-top:8px">
      <button class="btn btn-sm" id="bulkCopyBtn">Скопировать текст</button>
      <span class="small dim" id="bulkNoPhoneNote"></span>
    </div>
    <div id="bulkMsgList" style="margin-top:10px"></div>
  `);
  function renderList(){
    const text=$('#bulkMsgText').value;
    let noPhone=0;
    $('#bulkMsgList').innerHTML = clients.map(c=>{
      const digits=(c.whatsapp||c.phone||'').replace(/[^0-9]/g,'');
      const link = digits? `https://wa.me/${digits}?text=${encodeURIComponent(text)}` : '';
      if(!digits) noPhone++;
      return `<div class="row small" style="padding:7px 0;border-bottom:1px dashed var(--line)">
        <span>${esc(c.name)}</span>
        ${link?`<a class="btn btn-sm btn-primary" href="${link}" target="_blank" rel="noopener">WhatsApp</a>`:'<span class="dim">Нет номера</span>'}
      </div>`;
    }).join('');
    $('#bulkNoPhoneNote').textContent = noPhone? `Без номера: ${noPhone}` : '';
  }
  $('#bulkMsgText').oninput=renderList;
  $('#bulkCopyBtn').onclick=()=>{ navigator.clipboard?.writeText($('#bulkMsgText').value).then(()=>toast('Текст скопирован')).catch(()=>toast('Не удалось скопировать')); };
  renderList();
}
