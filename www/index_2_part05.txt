function cartridgeKey(u){ return u.config+'|'+(u.size||''); }
function renderCartridgeAnalysis(){
  const now=Date.now();
  function usageSince(days){
    const cutoff=now-days*86400000;
    const agg={};
    S.cartridgeUsage.filter(u=>new Date(u.date).getTime()>=cutoff).forEach(u=>{ const k=cartridgeKey(u); agg[k]=(agg[k]||0)+Number(u.qty||0); });

  }
  const u30=usageSince(30), u90=usageSince(90), u365=usageSince(365);

function renderPurchaseRecommendation(){
  const lead=Number(S.settings.leadTimeDays||7), reserve=Number(S.settings.reserve||3);
  const now=Date.now();
  const cutoff=now-30*86400000;
  const usage30={};
  S.cartridgeUsage.filter(u=>new Date(u.date).getTime()>=cutoff).forEach(u=>{ const k=cartridgeKey(u); usage30[k]=(usage30[k]||0)+Number(u.qty||0); });
  const cartridgeItems=invItemsByCat('cartridge');
  const rows=cartridgeItems.map(item=>{
    const k=cartridgeKey({config:item.config,size:item.size});
    const used30=usage30[k]||0;
    const hasHistory=used30>0;
    const avgDay=used30/30;
    const recommended = hasHistory ? Math.ceil(avgDay*lead+reserve) : null;
    const toBuy = hasHistory ? Math.max(0,recommended-item.qty) : null;
    const status = !hasHistory ? 'недостаточно истории' : (item.qty<=item.threshold?'докупить срочно':(toBuy>0?'докупить':'достаточно'));
    return {name:(item.config||'')+' '+(item.size||''),qty:item.qty,used30,avgDay,threshold:item.threshold,toBuy,status,hasHistory};
  });
  $('#invBody').innerHTML=`
  <div class="card">
    <h3>Рекомендация закупки картриджей</h3>
    <p class="small dim">Формула: рекомендуемый запас = средний расход в день × срок поставки (${lead} дн.) + резерв (${reserve} шт.). Если истории мало — используется минимальный остаток, и предлагается ввести количество вручную.</p>
    <div class="table-wrap"><table><tr><th>Конфигурация</th><th>Остаток</th><th>30 дн.</th><th>Средн./день</th><th>Мин.</th><th>Купить</th><th>Статус</th></tr>
    ${rows.length?rows.map(r=>`<tr style="${r.status.includes('срочно')?'background:rgba(198,43,40,.1)':''}">
      <td>${esc(r.name)}</td><td>${r.qty}</td><td>${r.used30}</td><td>${r.avgDay.toFixed(2)}</td><td>${r.threshold}</td>
      <td>${r.hasHistory? r.toBuy : '<span class="dim">вручную</span>'}</td><td>${r.status}</td></tr>`).join(''):'<tr><td colspan="7" class="dim">Нет картриджей на складе. Добавьте позиции в разделе «Картриджи».</td></tr>'}
    </table></div>
  </div>`;
}

RENDERERS.cities=function(){
  const agg={};
  S.clients.forEach(c=>{
    if(!c.city) return;
    agg[c.city]=agg[c.city]||{clients:0,revenue:0,visits:0};

  const startBtn=$('#camStartBtn');

function stopCamera(){ if(photoState.cameraStream){ photoState.cameraStream.getTracks().forEach(t=>t.stop()); photoState.cameraStream=null; } }
function analyzeFrameLoop(){
  const v=$('#camVideo'); const ov=$('#camOverlay');
  if(!v || v.style.display==='none' || !photoState.cameraStream) return;
  ov.width=v.clientWidth; ov.height=v.clientHeight;
  const tmp=document.createElement('canvas'); tmp.width=80; tmp.height=100;
  const tctx=tmp.getContext('2d'); tctx.drawImage(v,0,0,80,100);
  let data; try{ data=tctx.getImageData(0,0,80,100).data; }catch(e){ return; }
  let sum=0,count=0,edgeSum=0;
  for(let i=0;i<data.length;i+=4){ const l=(data[i]+data[i+1]+data[i+2])/3; sum+=l; count++; }
  const avg=sum/count; // brightness 0-255
  // crude sharpness via variance of luminance diff between neighboring samples
  let variance=0,prev=null,n=0;
  for(let i=0;i<data.length;i+=16){ const l=(data[i]+data[i+1]+data[i+2])/3; if(prev!==null){ variance+=Math.abs(l-prev); n++; } prev=l; }
  const sharpness=n?variance/n:0;
  let score=100, hints=[];
  if(avg<60){ score-=30; hints.push('Слишком темно — измените свет'); }
  else if(avg>200){ score-=25; hints.push('Пересвет — уберите блик или уменьшите свет'); }
  if(sharpness<3){ score-=25; hints.push('Возможна дрожь камеры — держите ровно'); }
  if(hints.length===0){ hints.push('Кадр выглядит стабильным'); }
  score=Math.max(0,Math.min(100,score));
  const frameEl=$('#cameraFrame');
  frameEl.className='camera-frame '+(score>70?'good':score>40?'warn':'bad');
  $('#camScoreWrap').innerHTML=`<div>Оценка кадра: <b>${score}/100</b></div><div class="dim">${hints.join(' · ')}</div>`;
  requestAnimationFrame(analyzeFrameLoop);
}

function applyPreset(key){
  const p=PHOTO_PRESETS[key];
  if($('#rBrightness')) $('#rBrightness').value=(p.brightness||1)*100;
  if($('#rContrast')) $('#rContrast').value=(p.contrast||1)*100;

function renderEditCanvas(){
  if(!photoState.img) return;
  const canvas=$('#editCanvas'); const img=photoState.img;
  let w=img.width,h=img.height; const maxW=640; if(w>maxW){ h=h*maxW/w; w=maxW; }
  canvas.width=w; canvas.height=h;
  const ctx=canvas.getContext('2d');
  const b=($('#rBrightness')?.value||100)/100, c=($('#rContrast')?.value||100)/100, s=($('#rSaturation')?.value||100)/100, r=($('#rRedReduce')?.value||0)/100;
  ctx.filter=`brightness(${b}) contrast(${c}) saturate(${s})`;
  ctx.drawImage(img,0,0,w,h);
  ctx.filter='none';
  if(r>0){
    const id=ctx.getImageData(0,0,w,h); const d=id.data;
    for(let i=0;i<d.length;i+=4){
      const avg=(d[i]+d[i+1]+d[i+2])/3;
      if(d[i]>d[i+1]+18 && d[i]>d[i+2]+18){ d[i]=d[i]-(d[i]-avg)*r; }
    }
    ctx.putImageData(id,0,0);
  }
  if($('#cLogo')?.checked && S.settings.logo){
    const logo=new Image();
    logo.onload=()=>{ const lw=w*0.22; const lh=logo.height*(lw/logo.width); ctx.globalAlpha=0.85; ctx.drawImage(logo,w-lw-10,h-lh-10,lw,lh); ctx.globalAlpha=1; };
    logo.src=S.settings.logo;
  }
}

function generateWeeklyReport(){
  const now=new Date(); const start=new Date(now); start.setDate(now.getDate()-now.getDay()+1);
  const s=start.toISOString().slice(0,10);
  const weekEvents=S.events.filter(e=>e.date>=s);
  const sessions=weekEvents.filter(e=>e.type==='session').length;
  const consults=weekEvents.filter(e=>e.type==='consult').length;
  const corrections=weekEvents.filter(e=>e.type==='correction').length;
  const models=weekEvents.filter(e=>e.type==='model').length;
  const newClients=S.clients.filter(c=>c.firstVisit>=s).length;
  const revenue=S.finance.filter(f=>f.type==='income'&&f.date>=s).reduce((a,f)=>a+Number(f.amount||0),0);
  const expenses=S.finance.filter(f=>f.type==='expense'&&f.date>=s).reduce((a,f)=>a+Number(f.amount||0),0);
  const filmSales=S.healingFilmSales.filter(f=>f.date>=s).reduce((a,f)=>a+Number(f.amount||0),0);
  const text=`Отчёт за неделю от ${fmtDate(s)}\n\nСеансы: ${sessions}\nКонсультации: ${consults}\nКоррекции: ${corrections}\nМодели: ${models}\nНовые клиенты: ${newClients}\nВыручка: ${fmtMoney(revenue)}\nРасходы: ${fmtMoney(expenses)}\nПродажи плёнки: ${fmtMoney(filmSales)}\nЧистая прибыль (предварительно): ${fmtMoney(revenue-expenses)}`;
  const rev={id:uid(),date:todayISO(),weekStart:s,text};
  idbPut('reminders',rev);
  window._lastReportText=text;
  window._lastReportSeries=[
    {label:'Сеансы',value:sessions,color:'#20B2AA'},
