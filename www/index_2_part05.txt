function openDaySummary(iso){

  const hours=dayEvs.reduce((s,e)=>s+eventHours(e),0);
  const workStart=S.settings.workHoursStart||10, workEnd=S.settings.workHoursEnd||20;
  const busyRanges=dayEvs.filter(e=>e.start&&e.end).map(e=>[e.start,e.end]).sort();
  let freeWindows=[]; let cursor=String(workStart).padStart(2,'0')+':00';

  const revenue=S.finance.filter(f=>f.type==='income'&&f.date===iso).reduce((s,f)=>s+Number(f.amount||0),0);

function findOverlappingEvents(date,start,end,excludeId){
  if(!date||!start||!end) return [];
  return S.events.filter(ev=>ev.id!==excludeId && ev.date===date && ev.status!=='cancelled' && ev.status!=='noshow' && ev.start && ev.end && (start<ev.end && end>ev.start));
}
async function finalizeEventSave(obj){

}

    // update client stats
    if(client){

      client.lastVisit=todayISO();

    }
    // after-session reminders explicit

    if(currentRoute==='calendar') RENDERERS.calendar();
  };

function periodRangeISO(period,offset){
  offset=offset||0;
  const now=new Date();
  let start,end;
  if(period==='day'){ start=new Date(now); start.setDate(start.getDate()+offset); end=new Date(start); }
  else if(period==='week'){ start=new Date(now); start.setDate(now.getDate()-((now.getDay()+6)%7)+offset*7); end=new Date(start); end.setDate(start.getDate()+6); }
  else if(period==='month'){ start=new Date(now.getFullYear(),now.getMonth()+offset,1); end=new Date(now.getFullYear(),now.getMonth()+offset+1,0); }
  else if(period==='quarter'){ const qAbs=Math.floor(now.getMonth()/3)+offset; const y=now.getFullYear()+Math.floor(qAbs/4); const qq=((qAbs%4)+4)%4; start=new Date(y,qq*3,1); end=new Date(y,qq*3+3,0); }
  else if(period==='year'){ start=new Date(now.getFullYear()+offset,0,1); end=new Date(now.getFullYear()+offset,11,31); }
  else { start=new Date(2000,0,1); end=new Date(now); }
  return {start:start.toISOString().slice(0,10),end:end.toISOString().slice(0,10)};
}function pctChange(cur,prev){
  if(!prev) return cur>0?100:0;
  return Math.round((cur-prev)/prev*100);
}
function trendHtml(pct){
  if(!isFinite(pct)) return '';
  const up=pct>=0;
  return `<span class="trend ${up?'up':'down'}">${up?'↗':'↘'} ${up?'+':''}${pct}%</span>`;
}
function buildIncomeSeries(startISO,endISO,granularity){
  const points=[]; let cur=new Date(startISO); const end=new Date(endISO);
  let guard=0;
  while(cur<=end && guard<400){
    guard++;
    let bucketEndDate;
    if(granularity==='day') bucketEndDate=new Date(cur);
    else if(granularity==='week'){ bucketEndDate=new Date(cur); bucketEndDate.setDate(bucketEndDate.getDate()+6); }
    else bucketEndDate=new Date(cur.getFullYear(),cur.getMonth()+1,0);
    const bStart=cur.toISOString().slice(0,10);
    const bEnd=(bucketEndDate>end?end:bucketEndDate).toISOString().slice(0,10);
    const val=S.finance.filter(f=>f.type==='income'&&f.date>=bStart&&f.date<=bEnd).reduce((s,f)=>s+Number(f.amount||0),0);
    const label=granularity==='month'? cur.toLocaleDateString('ru-RU',{month:'short'}) : String(cur.getDate());
    points.push({label,value:val});
    if(granularity==='day') cur.setDate(cur.getDate()+1);
    else if(granularity==='week') cur.setDate(cur.getDate()+7);
    else cur=new Date(cur.getFullYear(),cur.getMonth()+1,1);
  }
  return points;
}

function financeScope(f){
  if(!f.date) return financePeriod==='all';
  const now=new Date(); const d=new Date(f.date);
  if(financePeriod==='all') return true;
  if(financePeriod==='day') return f.date===todayISO();
  if(financePeriod==='week'){ const s=new Date(now); s.setDate(now.getDate()-now.getDay()+1); return d>=s; }
  if(financePeriod==='month') return f.date.slice(0,7)===now.toISOString().slice(0,7);
  if(financePeriod==='quarter'){ const q=Math.floor(now.getMonth()/3); return d.getFullYear()===now.getFullYear() && Math.floor(d.getMonth()/3)===q; }
  if(financePeriod==='year') return d.getFullYear()===now.getFullYear();
  return true;
}

async function saveSkills(studentId){
  const s=S.students.find(x=>x.id===studentId); if(!s) return;
  s.skills=s.skills||{};
  $$('#genericSheetBody input[type=range]').forEach(r=>{ s.skills[r.dataset.i]=Number(r.value); });
  await idbPut('students',s);
  closeSheet(); toast('Оценки сохранены');
  if(document.getElementById('studentDetailBody')) openStudent(studentId);
}
function skillsAvg(s){
  const vals=Object.values(s.skills||{});
  return vals.length? (vals.reduce((a,b)=>a+b,0)/vals.length) : 0;
}
